    <script>
        // --- 1. Password Protection Logic ---
        function checkPassword() {
            const input = document.getElementById('passcode').value;
            if(input === 'admin2026') {
                document.getElementById('lock-screen').style.display = 'none';
                document.getElementById('app-content').classList.remove('hidden');
                updateDashboard(); // Run initial calculation
            } else {
                document.getElementById('error-msg').classList.remove('hidden');
            }
        }

        // Allow 'Enter' key to submit password
        document.getElementById('passcode').addEventListener('keypress', function (e) {
            if (e.key === 'Enter') {
                checkPassword();
            }
        });

        // --- 2. State & UI Controls ---
        let state = {
            travelType: 'Overseas',
            milesBasis: 'Enrich (MH)'
        };

        const MILES_VAL = {
            'Enrich (MH)': 0.05, 
            'KrisFlyer (SQ)': 0.07, 
            'AirAsia BIG': 0.015
        };

        let chartInstance = null;

        function setTravelType(val, btn) {
            state.travelType = val;
            document.querySelectorAll('#travel-type-group .toggle-btn').forEach(b => b.classList.remove('active'));
            btn.classList.add('active');
            updateDashboard();
        }

        function setMilesBasis(val, btn) {
            state.milesBasis = val;
            document.querySelectorAll('#miles-basis-group .toggle-btn').forEach(b => b.classList.remove('active'));
            btn.classList.add('active');
            updateDashboard();
        }

        function switchTab(view, btn) {
            document.getElementById('view-table').classList.add('hidden');
            document.getElementById('view-chart').classList.add('hidden');
            document.getElementById('view-' + view).classList.remove('hidden');
            
            document.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('active'));
            btn.classList.add('active');
        }

        // Utility formatters
        const fm = (num) => 'RM ' + num.toLocaleString('en-MY', {minimumFractionDigits: 0, maximumFractionDigits: 0});
        const fmDec = (num) => 'RM ' + num.toLocaleString('en-MY', {minimumFractionDigits: 2, maximumFractionDigits: 2});

        // --- 3. The 11+1 Month Calculation Engine ---
        function updateDashboard() {
            // Read inputs
            const s = {
                petrol: parseInt(document.getElementById('sl-petrol').value),
                dining: parseInt(document.getElementById('sl-dining').value),
                grocery: parseInt(document.getElementById('sl-grocery').value),
                ent: parseInt(document.getElementById('sl-ent').value),
                ewallet: parseInt(document.getElementById('sl-ewallet').value),
                travel_mo: parseInt(document.getElementById('sl-travel-mo').value),
                travel_yr: parseInt(document.getElementById('sl-travel-yr').value),
                online: parseInt(document.getElementById('sl-online').value),
                other: parseInt(document.getElementById('sl-other').value)
            };

            // Update UI Labels
            document.getElementById('val-petrol').innerText = fm(s.petrol);
            document.getElementById('val-dining').innerText = fm(s.dining);
            document.getElementById('val-grocery').innerText = fm(s.grocery);
            document.getElementById('val-ent').innerText = fm(s.ent);
            document.getElementById('val-ewallet').innerText = fm(s.ewallet);
            document.getElementById('val-travel-mo').innerText = fm(s.travel_mo);
            document.getElementById('val-travel-yr').innerText = fm(s.travel_yr);
            document.getElementById('val-online').innerText = fm(s.online);
            document.getElementById('val-other').innerText = fm(s.other);

            // Base metrics
            const reg_mo_spend = s.petrol + s.dining + s.grocery + s.ent + s.ewallet + s.travel_mo + s.online + s.other;
            const peak_mo_spend = reg_mo_spend + s.travel_yr;
            const annual_spend = (reg_mo_spend * 11) + peak_mo_spend;
            const avg_mo_spend = annual_spend / 12;

            let cards = [];
            const unirmToMiles = (unirm) => unirm / 12;

            // 1. UOB Visa Infinite Metal
            const dining_mult = s.dining >= 1000 ? 5 : 1;
            const uobMetalMo = (t) => t * (state.travelType === 'Overseas' ? 10 : 1) + s.dining * dining_mult + s.grocery + s.ent + s.other + s.online + s.petrol + s.ewallet;
            let uob_metal_earn = (uobMetalMo(s.travel_mo) * 11) + uobMetalMo(s.travel_mo + s.travel_yr);
            let uob_metal_miles = unirmToMiles(uob_metal_earn + 300000);
            cards.push({ name: 'UOB VI Metal', type: 'Miles', color: '#0A3B72', miles: uob_metal_miles, fee: 3000, remark: '300k Bonus, Limo' });

            // 2. UOB Visa Infinite
            let uob_vi_earn = (uobMetalMo(s.travel_mo) * 11) + uobMetalMo(s.travel_mo + s.travel_yr);
            let uob_vi_miles = unirmToMiles(uob_vi_earn);
            cards.push({ name: 'UOB Visa Infinite', type: 'Miles', color: '#185FA5', miles: uob_vi_miles, fee: annual_spend >= 50000 ? 0 : 600, remark: '12X Lounge' });

            // 3. UOB PRVI Miles Elite
            const overseas_mult = state.travelType === 'Overseas' ? 12 : 5;
            const uobPmeMo = (t) => t * overseas_mult + s.dining + s.grocery + s.ent + s.other + s.online + s.petrol + s.ewallet;
            let uob_pme_fee = annual_spend >= 50000 ? 0 : 600;
            let uob_pme_bonus = uob_pme_fee === 0 ? 0 : 60000;
            let uob_pme_earn = (uobPmeMo(s.travel_mo) * 11) + uobPmeMo(s.travel_mo + s.travel_yr);
            cards.push({ name: 'UOB PRVI Elite', type: 'Miles', color: '#0F6E56', miles: unirmToMiles(uob_pme_earn + uob_pme_bonus), fee: uob_pme_fee, remark: '8X Lounge' });

            // 4. UOB PRVI Miles
            const uobPrviMo = (t) => {
                let e_5x = Math.min(t, 5000);
                let e_1x = t - e_5x;
                return (e_5x * 5) + e_1x + s.dining + s.grocery + s.ent + s.other + s.online + s.petrol + s.ewallet;
            };
            let uob_prvi_earn = (uobPrviMo(s.travel_mo) * 11) + uobPrviMo(s.travel_mo + s.travel_yr);
            cards.push({ name: 'UOB PRVI Miles', type: 'Miles', color: '#34D399', miles: unirmToMiles(uob_prvi_earn), fee: annual_spend >= 20000 ? 0 : 198, remark: '50% Grab' });

            // 5 & 6. UOB ONE
            const uobOneMo = (t, isPlat) => {
                let mo_s = s.petrol + s.dining + s.grocery + s.ent + s.ewallet + t + s.online + s.other;
                let min_s = isPlat ? 1500 : 800;
                let cap = isPlat ? 15 : 10;
                if (mo_s >= min_s) {
                    let p = Math.min(s.petrol * 0.1, cap); let d = Math.min(s.dining * 0.1, cap);
                    let g = Math.min(s.grocery * 0.1, cap); let e = Math.min(s.ewallet * 0.1, cap);
                    let b = (mo_s - s.petrol - s.grocery - s.dining - s.ewallet) * 0.002;
                    return p + d + g + e + b;
                }
                return mo_s * 0.002;
            };
            cards.push({ name: 'UOB ONE Platinum', type: 'Cashback', color: '#F59E0B', cb: (uobOneMo(s.travel_mo, true)*11) + uobOneMo(s.travel_mo+s.travel_yr, true), fee: annual_spend >= 20000 ? 0 : 195, remark: 'Max RM15/cat' });
            cards.push({ name: 'UOB ONE Classic', type: 'Cashback', color: '#FCD34D', cb: (uobOneMo(s.travel_mo, false)*11) + uobOneMo(s.travel_mo+s.travel_yr, false), fee: annual_spend >= 15000 ? 0 : 120, remark: 'Max RM10/cat' });

            // 7. HSBC TravelOne
            const hsbcMo = (t) => t * (state.travelType === 'Overseas' ? 8 : 5) + s.dining * 5 + s.grocery + s.ent + s.other + s.online;
            let hsbc_div = state.milesBasis === 'Enrich (MH)' ? 24 : state.milesBasis === 'KrisFlyer (SQ)' ? 30 : 25;
            cards.push({ name: 'HSBC TravelOne', type: 'Miles', color: '#D85A30', miles: ((hsbcMo(s.travel_mo)*11) + hsbcMo(s.travel_mo+s.travel_yr)) / hsbc_div, fee: annual_spend >= 20000 ? 0 : 300, remark: '6X Lounge' });

            // 8. SC Journey
            const scMo = (t) => t * (state.travelType === 'Overseas' ? 1 : 0.2) + s.dining + s.ent*0.2 + s.grocery*0.2 + s.other*0.2 + s.online*0.2;
            cards.push({ name: 'SC Journey', type: 'Miles', color: '#7F77DD', miles: ((scMo(s.travel_mo)*11) + scMo(s.travel_mo+s.travel_yr)) / 2, fee: annual_spend >= 60000 ? 0 : 600, remark: 'Unl. Lounge KLIA' });

            // 9 & 10. CIMB Travel
            let cimb_div = state.milesBasis === 'Enrich (MH)' ? 12.5 : state.milesBasis === 'KrisFlyer (SQ)' ? 15 : 8;
            const cimbTpMo = (t) => t * (state.travelType === 'Overseas' ? 5 : 2) + (s.dining+s.grocery+s.ent+s.other+s.online)*2 + s.petrol + s.ewallet;
            cards.push({ name: 'CIMB Travel Plat', type: 'Miles', color: '#A8282D', miles: ((cimbTpMo(s.travel_mo)*11) + cimbTpMo(s.travel_mo+s.travel_yr)) / cimb_div, fee: 0, remark: 'Free for life' });
            
            const cimbTwMo = (t) => t * (state.travelType === 'Overseas' ? 8 : 2) + (s.dining+s.grocery+s.ent+s.other+s.online)*2 + s.petrol + s.ewallet;
            cards.push({ name: 'CIMB Travel World', type: 'Miles', color: '#8C1F23', miles: ((cimbTwMo(s.travel_mo)*11) + cimbTwMo(s.travel_mo+s.travel_yr)) / cimb_div, fee: annual_spend >= 100000 ? 0 : 555, remark: '12X Lounge' });

            // 11. CIMB VI
            const cimbViMo = (t) => t * (state.travelType === 'Overseas' ? 5 : 1) + s.dining*5 + s.online*5 + s.grocery + s.ent + s.other + s.petrol + s.ewallet;
            cards.push({ name: 'CIMB Visa Infinite', type: 'Miles', color: '#C73B41', miles: ((cimbViMo(s.travel_mo)*11) + cimbViMo(s.travel_mo+s.travel_yr)) / cimb_div, fee: 0, remark: '8X Lounge' });

            // 12. CIMB Cash Rebate
            const cimbCrMo = (t) => {
                let mo = s.petrol + s.dining + s.grocery + s.ent + s.ewallet + t + s.online + s.other;
                if (mo >= 3000) return Math.min((s.grocery+s.petrol+s.ent)*0.05, 30) + (s.dining+s.ewallet+t+s.online+s.other)*0.002;
                return mo * 0.002;
            };
            cards.push({ name: 'CIMB Cash Rebate', type: 'Cashback', color: '#D95257', cb: (cimbCrMo(s.travel_mo)*11) + cimbCrMo(s.travel_mo+s.travel_yr), fee: 0, remark: 'Cap RM30' });

            // 13. CIMB Petronas
            const cimbPetMo = (t) => {
                let mo = s.petrol + s.dining + s.grocery + s.ent + s.ewallet + t + s.online + s.other;
                if (mo >= 1500) return Math.min(s.petrol*0.07, 30) + (mo - s.petrol)*0.002;
                return mo * 0.002;
            };
            cards.push({ name: 'CIMB PETRONAS-i', type: 'Cashback', color: '#009F75', cb: (cimbPetMo(s.travel_mo)*11) + cimbPetMo(s.travel_mo+s.travel_yr), fee: 0, remark: 'Cap RM30 Setel' });

            // 14. CIMB e Credit
            const cimbEMo = (t) => s.ewallet*8 + s.online*8 + s.petrol + s.dining + s.grocery + s.ent + t + s.other;
            cards.push({ name: 'CIMB e Card', type: 'Miles', color: '#E67B7F', miles: ((cimbEMo(s.travel_mo)*11) + cimbEMo(s.travel_mo+s.travel_yr)) / cimb_div, fee: annual_spend >= 6000 ? 0 : 80, remark: '10X eDay' });

            // 15. RHB World
            const rhbMo = (t) => {
                let mo = s.petrol + s.dining + s.grocery + s.ent + s.ewallet + t + s.online + s.other;
                let rate = mo >= 3500 ? 0.06 : mo >= 2500 ? 0.02 : mo >= 1000 ? 0.01 : 0.002;
                let c_cats = (s.dining + t + s.petrol) * rate;
                let o_cats = (s.grocery + s.ent + s.online + s.other + s.ewallet) * 0.002;
                return Math.min(c_cats, 30) + Math.min(o_cats, 30);
            };
            cards.push({ name: 'RHB World', type: 'Cashback', color: '#639922', cb: (rhbMo(s.travel_mo)*11) + rhbMo(s.travel_mo+s.travel_yr), fee: 0, remark: 'Up to 6% Tiered' });


            // --- Process Results ---
            cards = cards.map(c => {
                let rmVal = c.type === 'Miles' ? (c.miles * MILES_VAL[state.milesBasis]) : c.cb;
                c.rmVal = rmVal;
                c.net = rmVal - c.fee;
                return c;
            }).sort((a,b) => b.net - a.net);

            const bestMiles = cards.find(c => c.type === 'Miles');
            const bestCash = cards.find(c => c.type === 'Cashback');

            // --- Update Top Metrics ---
            const w_stat = annual_spend >= 50000 ? '<span class="text-green-400">Yes ✓</span>' : `<span class="text-amber-400">No &mdash; need ${fm(50000 - annual_spend)} more</span>`;
            document.getElementById('metrics-container').innerHTML = `
                <div class="bg-gray-900/50 p-4 rounded-lg border border-brand-border">
                    <div class="text-brand-muted text-xs uppercase tracking-wide mb-1">Avg Monthly Spend</div>
                    <div class="text-2xl font-bold">${fm(avg_mo_spend)}</div>
                    <div class="text-brand-muted text-xs mt-1">${fm(annual_spend)} / yr</div>
                </div>
                <div class="bg-gray-900/50 p-4 rounded-lg border border-brand-border">
                    <div class="text-brand-muted text-xs uppercase tracking-wide mb-1">Fee Waiver (UOB VI)</div>
                    <div class="text-lg font-bold leading-tight mt-1">${w_stat}</div>
                    <div class="text-brand-muted text-xs mt-1">RM50k threshold</div>
                </div>
                <div class="bg-gray-900/50 p-4 rounded-lg border border-brand-border ring-1 ring-blue-500/30 shadow-[0_0_15px_rgba(59,130,246,0.15)]">
                    <div class="text-blue-300 text-xs uppercase tracking-wide mb-1">Best Miles Card</div>
                    <div class="text-2xl font-bold">${fmDec(bestMiles.net)}</div>
                    <div class="text-brand-muted text-xs mt-1 truncate">${bestMiles.name}</div>
                </div>
                <div class="bg-gray-900/50 p-4 rounded-lg border border-brand-border ring-1 ring-green-500/30 shadow-[0_0_15px_rgba(34,197,94,0.15)]">
                    <div class="text-green-300 text-xs uppercase tracking-wide mb-1">Best Cashback</div>
                    <div class="text-2xl font-bold">${fmDec(bestCash.net)}</div>
                    <div class="text-brand-muted text-xs mt-1 truncate">${bestCash.name}</div>
                </div>
            `;

            // --- Update Table ---
            let tbody = '';
            cards.forEach(c => {
                let mText = c.type === 'Miles' ? Math.round(c.miles).toLocaleString() + ' mi' : '&mdash;';
                let fText = c.fee === 0 ? '<span class="text-green-400">Waived / Free</span>' : `<span class="text-red-400">-RM ${c.fee}</span>`;
                let nColor = c.net >= 0 ? 'text-white' : 'text-red-400';
                
                tbody += `
                    <tr class="hover:bg-brand-dark/50 transition-colors">
                        <td class="px-4 py-3 font-medium">
                            ${c.name}
                            <div class="text-[10px] text-brand-muted mt-0.5 whitespace-normal w-48">${c.remark}</div>
                        </td>
                        <td class="px-4 py-3 text-xs"><span class="px-2 py-1 rounded-full bg-brand-dark border border-brand-border">${c.type}</span></td>
                        <td class="px-4 py-3 text-right font-mono text-sm text-gray-300">${mText}</td>
                        <td class="px-4 py-3 text-right font-mono text-sm">${fmDec(c.rmVal)}</td>
                        <td class="px-4 py-3 text-right font-mono text-sm">${fText}</td>
                        <td class="px-4 py-3 text-right font-mono font-bold ${nColor}">${fmDec(c.net)}</td>
                    </tr>
                `;
            });
            document.getElementById('table-body').innerHTML = tbody;

            // --- Update Chart ---
            renderChart(cards);
        }

        function renderChart(cards) {
            const ctx = document.getElementById('barChart').getContext('2d');
            
            const labels = cards.map(c => c.name.replace('UOB ', '').replace('CIMB ', '').replace('RHB ', ''));
            const data = cards.map(c => c.net);
            const colors = cards.map(c => c.color);

            if (chartInstance) {
                chartInstance.destroy();
            }

            Chart.defaults.color = '#94a3b8';
            Chart.defaults.font.family = 'system-ui, sans-serif';

            chartInstance = new Chart(ctx, {
                type: 'bar',
                data: {
                    labels: labels,
                    datasets: [{
                        label: 'Net Annual Value (RM)',
                        data: data,
                        backgroundColor: colors,
                        borderRadius: 4,
                        borderSkipped: false
                    }]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    plugins: {
                        legend: { display: false },
                        tooltip: {
                            backgroundColor: '#1e293b',
                            titleColor: '#f8fafc',
                            bodyColor: '#f8fafc',
                            borderColor: '#334155',
                            borderWidth: 1,
                            callbacks: {
                                label: function(context) {
                                    return ' RM ' + context.parsed.y.toFixed(2);
                                }
                            }
                        }
                    },
                    scales: {
                        y: {
                            grid: { color: '#334155', tickLength: 0 },
                            border: { display: false },
                            ticks: {
                                callback: function(value) { return 'RM ' + value; },
                                padding: 10
                            }
                        },
                        x: {
                            grid: { display: false },
                            border: { display: false },
                            ticks: { maxRotation: 45, minRotation: 45 }
                        }
                    }
                }
            });
        }
